% AutoBuy -- automated crypto purchasing against customers' own exchange accounts.
\begin{cvproject}{AutoBuy}{Python, FastAPI, PostgreSQL, Redis, React}
  \item Automated crypto purchasing platform that trades through each customer's own exchange account.
  \item Migrated a \textbf{12-detector} account-monitoring engine from SQLite to PostgreSQL, covering \textbf{Bitget, Bybit, OKX, and Binance} with read-only, trade-incapable credentials.
  \item Shipped MFA (authenticator codes, recovery codes, global session revocation) and a \textbf{600+ test} suite for the monitoring engine.
\end{cvproject}
